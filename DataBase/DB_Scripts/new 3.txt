-- DROP FUNCTION freightsacclive.rcti_invoicegenerateorpreview_s(int8, int8, date, int8, int2);

CREATE OR REPLACE FUNCTION freightsacclive.rcti_invoicegenerateorpreview_s(p_fleetownerid bigint, p_branchid bigint, p_dateto date, p_refuser bigint, p_generateinvoice smallint DEFAULT 0)
 RETURNS TABLE(invoiceid bigint, invoiceno character varying, reffleetowner bigint, refbranch bigint, accountno character varying, fleetownername character varying, baserateexgst money, extrachargeexgst money, totalexgst money, totalgst money, gstvalue money, total money, datefrom date, dateto date, refinvoicestatus bigint, invoicestatusdate timestamp without time zone, totalpaymentreceived money, comments character varying, createdate timestamp without time zone, refusercreatedby bigint, updatedate timestamp without time zone, refuser bigint, invoicefrequency character varying, docketids character varying)
 LANGUAGE plpgsql
AS $function$
declare
	fleetownerrecord 	record;
	docketrecord 		record;
	newInvoiceRecord 		record;
	fleetownerraterecord record;
	jobchargerecord		record;
    periodicchargerecord record;
	v_datefrom 	date;
	v_dateto 	date;
	v_oldestdocketdate 	date; --this will help to cut invoice ranges because we do not need to check for invoices before this date--		
	v_branch 	bigint;
	v_docketstatusverified 		docketstatus.id%type;  		--verified  = 8-- 
	v_docketstatusrctiissued 	docketstatus.id%type; 		--cancelled = 9--
	v_fleetownerratetypezonerate			 	fleetownerratetype.id%type;	-- zonerate = 1 --
	v_fleetownerratetypepercentofclientrate		fleetownerratetype.id%type;	-- percentofclientrate = 2 --
	v_fleetownerratetypehourlyrate		 		fleetownerratetype.id%type;	-- hourlyrate = 3 --
	v_FleetOwnerRateTypeFixedeRate		 		fleetownerratetype.id%type;	-- hourlyrate = 4 --
	v_docketcount 				bigint;
	v_gstvalue 					numeric;
	v_periodicchargetype 		bigint;
	v_invoicechargevalue 		numeric;
	v_baserateexgst 			numeric;
	v_extrachargeexgst 			numeric;
	v_totalexgst 				numeric;
	v_totalgst 					numeric;
	v_total 					numeric;
	v_initialinvoicedate		date;
	v_lastinvoicedate			date;
	v_description				varchar;
	v_newinvoicenumber			varchar;
	arr_invoicetodates			date[];
	max_invoicetodate			date;
	debugvariable				varchar;
	v_invoicepersisted			smallint;
	v_docket_docketpayment		money;
	v_docket_extrachargeexgst	money;
	v_dockettotalexgst			money;
	v_dockettotalgst			money;
	v_debitdocumentRCTI	bigint;
	v_debitdocumentDeductionTaxInvoice	bigint;
	v_docketchargedescription  varchar;
	v_completeddocketitems      int=0;
	v_completedtotaldocketitems int=0;
	v_deliveryjobtype       jobtype.id%type := 2;
	v_totaldocumentpayment		numeric;
	v_rcti_invoiceitemexists    smallint;
	v_Rcti_InvoiceChargeExists  smallint;
	v_TotalForPeriodicCharges    numeric := 0.0;
	v_PeriodicChargeValue       numeric := 0.0;
	v_DeductionInvoiceId    	bigint;
	v_DeductionInvoiceTotalExGst numeric;
	v_DeductionInvoiceTotalGst   numeric;
    v_count            bigint;


begin
	v_dateto := p_dateto;
	v_docketstatusverified		:= 8;
	v_docketstatusrctiissued	:= 9;
	v_fleetownerratetypezonerate			:= 1;
	v_fleetownerratetypepercentofclientrate	:= 2;
	v_fleetownerratetypehourlyrate		 	:= 3;	
    v_FleetOwnerRateTypeFixedeRate          := 4;
	v_debitdocumentRCTI	:= 4;
    v_debitdocumentDeductionTaxInvoice := 5;
	v_branch 				:= p_branchid;
	v_gstvalue 				:= 0.0;
	v_periodicchargetype 	:= 2;
	v_invoicechargevalue	:= 0; 
	v_invoicepersisted		:= 0; 
    

	if(v_dateto > now()) then v_dateto := now(); end if; 

	select coalesce(p.parameter_value, '2023-01-01')::date into v_initialinvoicedate from parameters p 
	where p.name = 'INVOICE.INITIAL_START_DATE' and p.refuser is null limit 1;
	
	select coalesce(p.parameter_value, '0.0')::numeric into v_gstvalue from parameters p 
	where p.name = 'INVOICE.GSTVALUE' and p.refuser is null  limit 1;
	

	for fleetownerrecord in 
		select fo.*, ifr.description invoicefrequency
		from fleetowner fo 
		inner join invoicefrequency ifr on fo.refinvoicefrequency = ifr.id
		where (p_fleetownerid is null or fo.id = p_fleetownerid) and fo.invoicestartdate <= v_dateto 
	loop --fleetownerrecord--
		perform 1
		from fleetowner
		where fleetowner.id=fleetownerrecord.id
		for update;
		arr_invoicetodates:=array[]::date[];
		max_invoicetodate:= null;
		Raise notice 'Fleetowner processing=% PID=%',fleetownerrecord.id,pg_backend_pid();
		if fleetownerrecord.invoicestartdate is null then continue; end if;-- no docket to be invoiced

		select coalesce(min(d.duedate), v_initialinvoicedate)::date into v_oldestdocketdate 
		from docket d 
		inner join run r on r.id = d.refrun 
		inner join driver dr on dr.id = r.refdriver and dr.reffleetowner = fleetownerrecord.id 
		where (d.refdocketstatus = v_docketstatusrctiissued) and (d.duedate between v_initialinvoicedate and v_dateto);
	    Raise notice 'run proceed';
		select coalesce( max(inv.dateto), v_initialinvoicedate) into v_lastinvoicedate
		from rcti_invoice inv 
		where inv.reffleetowner = fleetownerrecord.id  and inv.refdebitdocumenttype = v_debitdocumentRCTI; -- 1
		Raise notice 'last invoice date =%',v_lastinvoicedate;
		for newInvoiceRecord in (
			select startdate , invoicedate 
			from invoicerangebreakup_s(fleetownerrecord.invoicestartdate, v_dateto, v_initialinvoicedate, fleetownerrecord.refinvoicefrequency::smallint) 
			where (((startdate is null or startdate >= v_oldestdocketdate) and invoicedate >= v_oldestdocketdate) 
						or  (v_oldestdocketdate is null or invoicedate >= v_oldestdocketdate)) 
			  and (startdate >= v_lastinvoicedate)
				) --no need to invoice if already invoiced on this date range--
		-- only those invoiec ranges which can capture minimum docket date to be invoiced, so we do not check extra ranges--
		loop --newInvoiceRecord-- 
			v_invoicepersisted:=0;
			v_completedtotaldocketitems:=0;
			raise notice 'run proceed for fleetowner=%',fleetownerrecord.id;
		
			if((select count(unnest) from unnest(arr_invoicetodates) where unnest =newInvoiceRecord.invoicedate)<=0) then
				arr_invoicetodates:= array_append(arr_invoicetodates,newInvoiceRecord.invoicedate);
			end if;

			select max(unnest) into max_invoicetodate from unnest(arr_invoicetodates)
			where unnest < newInvoiceRecord.startdate;
	
     
			invoiceno := to_char(newInvoiceRecord.startdate, 'ddmmyy') || '' || to_char(newInvoiceRecord.invoicedate, 'ddmmyy') || 'R'; -- this is ok for preview, to generate actual invoice there should be some pattern
			reffleetowner := fleetownerrecord.id;
			refbranch := v_branch;
			refinvoicestatus :=1 ; --new --
			accountno := fleetownerrecord.code; 
			fleetownername := fleetownerrecord.description;
			baserateexgst :=  0::money;
			extrachargeexgst :=0::money;
			totalexgst := 0::money;
			totalgst :=0::money;
			total :=0::money;
			gstvalue := v_gstvalue;
			datefrom := newInvoiceRecord.startdate;
			dateto := newInvoiceRecord.invoicedate;
			refusercreatedby := p_refuser;
			updatedate := now();
			createdate := now();
			invoicestatusdate := now();
			refuser := p_refuser;
			invoicefrequency := fleetownerrecord.invoicefrequency;
			docketids := '~';
		
			--this line will determine if there any docket exist to be invoiced.
			select count(d.id) , sum (d.docketpayment::numeric)  
			into v_docketcount , v_totaldocumentpayment 
			from docket d  
			inner join run r on r.id= d.refrun 
			inner join driver dr on dr.id= r.refdriver and dr.reffleetowner = fleetownerrecord.id			
			where d.refdocketstatus = v_docketstatusverified
			  and d.duedate <= newInvoiceRecord.invoicedate
			  and (p_generateinvoice = 1
 				or (
					p_generateinvoice = 0 
					and (max_invoicetodate is null or d.duedate > max_invoicetodate )
					and d.duedate <= newInvoiceRecord.invoicedate
				)
			  );
					 
			if (coalesce(v_docketcount, 0) = 0  or coalesce(v_totaldocumentpayment,0.0) = 0) then
                raise notice 'skipped';
				continue;
			end if; -- do not invoice							
	
			if (p_generateinvoice = 1) then 
				invoiceid := nextval('RCTI_INVOICE_ID_SEQ');
			    invoiceno := nextval('INVOICE_NUMBER_SEQ')::varchar;
				insert into rcti_debug (createtime, sessionid ,fleetowner,msg )values (clock_timestamp(),pg_backend_pid(),fleetownerrecord.id,'before exists');
   				if exists(
					select 1
					from rcti_invoice ri
					where ri.reffleetowner = fleetownerrecord.id
					and ri.datefrom=newInvoiceRecord.startdate
					and ri.dateto= newInvoiceRecord.invoicedate
					and ri.refdebitdocumenttype=v_debitdocumentRCTI
	 				) then
						insert into rcti_debug (createtime, sessionid ,fleetowner,msg )values (clock_timestamp(),pg_backend_pid(),fleetownerrecord.id,'rcti_invoice already exist');	
						continue;
 				end if;
				insert into rcti_debug (createtime, sessionid ,fleetowner,msg )values (clock_timestamp(),pg_backend_pid(),fleetownerrecord.id,'after exists');	
				insert into rcti_debug (createtime, sessionid ,fleetowner,msg )values (clock_timestamp(),pg_backend_pid(),fleetownerrecord.id,'before insert');
				insert into rcti_invoice(
					  id
				  	, invoiceno			    
					, reffleetowner
					, refbranch
					, accountno
					, fleetownername
					, gstvalue
					, datefrom
					, dateto
					, refinvoicestatus
					, invoicestatusdate
					, createdate
					, refusercreatedby
					, updatedate
					, refuser
					, refdebitdocumenttype
					, creditmultiplier
					)
				values(
					  invoiceid
					, invoiceno ||'R'
					, reffleetowner
					, refbranch 
					, accountno 
					, fleetownername
					, gstvalue 
					, datefrom 
					, dateto 
					, refinvoicestatus 
					, invoicestatusdate 
					, createdate 
					, refusercreatedby 
					, updatedate 
					, refuser
					, v_debitdocumentRCTI
					, 1
				);		   
				
				v_invoicepersisted := 1;	
				v_DeductionInvoiceId := 0;
			end if;
			
			v_rcti_invoiceitemexists := 0;
		    v_Rcti_InvoiceChargeExists := 0;
			--if v_docketcount >= 1 then
			v_count :=0;
            select count(*) into v_count
			from rcti_invoice ri
			where ri.reffleetowner = fleetownerrecord.id
				and ri.datefrom=newInvoiceRecord.startdate
				and ri.dateto= newInvoiceRecord.invoicedate
				and ri.refdebitdocumenttype=v_debitdocumentRCTI;
			if(v_count>=2) then
				delete from rcti_invoice  where id = invoiceid;
				raise notice 'count become 2 so deleted one last record for session id=%',pg_backend_pid();
				continue;
			end if;
			for docketrecord in 
				select 
						d.id
					, d.refbooking
					, d.reffleetownerratetype
					, r.refdriver
					, d.duedate
					, b.distance
					--, r.refvehicletype
					--, coalesce(dip.qtyplan, 0) as qtyplan	
					, d.docketpayment
					, 0::money as extrachargeexgst
					, sp.description as fromsuburb
					, sd.description as tosuburb
					, sp.id as refsuburbfrom
					, sd.id as refsuburbto
					, b.reference1
					, '' as jobchargedescription
					, f.description as ratetypedescription
					, d.reffleetownerratetype as refratetype
					, lp.description as startlocationdescription
					, (coalesce(lp.addrline1,'') || ' ' || coalesce(lp.addrline2,''))::varchar  as startlocationaddress 
					, ld.description as endlocationdescription
					, (coalesce(ld.addrline1,'') || ' '  || coalesce(ld.addrline2,''))::varchar  as endlocationaddress 
					, d.basedocketrate 
					, d.totalhours
					, d.hourlycardrate

				from docket d  
				inner join run r on r.id= d.refrun 
				inner join driver dr on dr.id= r.refdriver and dr.reffleetowner = fleetownerrecord.id
				inner join booking b on b.id = d.refbooking
				inner join  fleetownerratetype f on d.reffleetownerratetype= f.id 			
				left join job jp on jp.refdocket = d.id and  jp.refjob_type = 1 -- pickup
				left join location lp on lp.id = jp.reflocation
				left join suburb sp on sp.id = lp.refsuburb
				left join job jd on jd.refdocket = d.id and  jd.refjob_type = 2 -- drop
				left join location ld on ld.id = jd.reflocation
				left join suburb sd on sd.id = ld.refsuburb			
				where d.refdocketstatus = v_docketstatusverified
				  and d.refbranch=v_branch 
				  and coalesce(d.docketpayment::numeric,0.0) > 0 
				  and d.duedate <= newInvoiceRecord.invoicedate
				  and (p_generateinvoice = 1
						or (
							p_generateinvoice = 0 
							and (max_invoicetodate is null or d.duedate > max_invoicetodate )
							and d.duedate <= newInvoiceRecord.invoicedate
							)
	  				  )
			loop	--docketrecord--			
				baserateexgst := baserateexgst + docketrecord.docketpayment;
				v_completeddocketitems := 0;			
				
				--  calculating job charges --				
				select string_agg(jobextracharges.chargedescription || ': ' || jobextracharges.charge::money::text, chr(13))
 				  ,sum(jobextracharges.charge)
				into docketrecord.jobchargedescription,	docketrecord.extrachargeexgst
				from (
					select chx.description chargedescription, calculatecharge(cc.code, jc.qty, d.docketpayment::numeric, jc.drivervalue) charge
					from job_charge  jc	
					inner join job j on jc.refjob = j.id 
					inner join docket d on d.id = docketrecord.id and j.refdocket = d.id
					inner join charge chx on chx.id=jc.refcharge 
					inner join chargecalculation cc on cc.id = chx.refchargecalculation
					where jc.drivervalue > 0 
					  and jc.isactive = 1 
					  and jc.refuserapprovedby is not null 
					  and jc.dateapprovedon is not null
					) jobextracharges
				where jobextracharges.charge > 0; -- only positive charges

				extrachargeexgst := extrachargeexgst + coalesce(docketrecord.extrachargeexgst, 0::money);
				docketids := docketids || docketrecord.id || '~';
				
				if (p_generateinvoice = 1 and v_invoicepersisted = 1) then				
					v_docketchargedescription := case when coalesce(docketrecord.extrachargeexgst, 0::money) > 0::money then docketrecord.jobchargedescription else null end;
					
					v_description := 
						case when docketrecord.refratetype = v_fleetownerratetypezonerate then
							'Charge Rate: ' || coalesce(docketrecord.basedocketrate,0::money)::varchar
						when docketrecord.refratetype = v_fleetownerratetypepercentofclientrate then
							'Percentage Rate: ' || coalesce(docketrecord.basedocketrate,0::money)::varchar
						when docketrecord.refratetype = v_fleetownerratetypehourlyrate then 
							'Hourly Rate: ' || coalesce(docketrecord.hourlycardrate,0::money)::varchar || chr(13) || 'hours: ' || docketrecord.totalhours						
						when docketrecord.refratetype = v_FleetOwnerRateTypeFixedeRate then
							'Fixed Rate: ' || coalesce(docketrecord.basedocketrate,0::money)::varchar							
						else ''
						end;
				
					if( fleetownerrecord.gstfree = 0) then 
						v_dockettotalgst := (docketrecord.docketpayment + coalesce(docketrecord.extrachargeexgst, 0::money)) * v_gstvalue::numeric/ 100.00::numeric;
					else
						v_dockettotalgst := 0::money;
					end if;
					
					select count(d.id) into v_completeddocketitems 
					from docket d
					inner join job j on j.refdocket = d.id and j.refjob_type=v_deliveryjobtype 
					inner join jobitemresult jir on jir.refjob = j.id and jir.completiondate is not null  
					inner join docketitem di on jir.refdocketitem = di.id
					where  d.id = docketrecord.id; 
					
					v_completedtotaldocketitems := v_completedtotaldocketitems + v_completeddocketitems;

					insert into rcti_invoiceitem(
						  refrctiinvoice
						, refbooking
						, refdocket
						, refdriver
						, reference1
						, refsuburbfrom
						, refsuburbto
						, baserateexgst
						, extrachargeexgst
						, totalexgst
						, totalgst
						, total
						, createdate
						, refusercreatedby
						, updatedate
						, refuser
						, description
						, itemdate
						,fromlocation
						,tolocation
						,extrachargedescription
						,totalitems
						)
					values (
						  invoiceid
						, docketrecord.refbooking
						, docketrecord.id
						, docketrecord.refdriver
						, docketrecord.reference1
						, docketrecord.refsuburbfrom
						, docketrecord.refsuburbto
						, docketrecord.docketpayment
						, docketrecord.extrachargeexgst
						, docketrecord.docketpayment + coalesce(docketrecord.extrachargeexgst, 0::money)	-- totalexgst
						, v_dockettotalgst::money															-- totalgst
						, docketrecord.docketpayment + coalesce(docketrecord.extrachargeexgst, 0::money) + v_dockettotalgst	-- total
						, now()
						, p_refuser
						, now()
						, p_refuser
						, v_description
						, docketrecord.duedate
						, docketrecord.startlocationdescription
						, docketrecord.endlocationdescription
						, v_docketchargedescription
						, v_completeddocketitems
					);
					
					v_rcti_invoiceitemexists := 1;
					
/*					for jobchargerecord in 	 
					    select jobextracharges.* 	from (
							select 
							chx.code, 
							chx.description chargedescription, 
							jc.createdate, 
							calculatecharge(cc.code, jc.qty, d.docketpayment::numeric,jc.drivervalue::numeric)::money charge						 
							from job_charge  jc	
							inner join job j on jc.refjob = j.id 
							inner join docket d on d.id = docketrecord.id and j.refdocket = d.id
							inner join charge chx on chx.id=jc.refcharge 
							inner join chargecalculation cc on cc.id = chx.refchargecalculation
							where jc.isactive = 1 
								and jc.refuserapprovedby is not null 
								and jc.dateapprovedon is not null
						    ) jobextracharges 
					    where charge::numeric < 0.0 ::numeric -- only positive charges				
					loop  --job jobchargerecord--				
						extrachargeexgst := extrachargeexgst + coalesce(jobchargerecord.charge, 0::money);							
						insert into rcti_invoiceitem(
							  refrctiinvoice
							, refbooking
							, refdocket
							, refdriver
							, reference1
							, refsuburbfrom
							, refsuburbto
							, baserateexgst
							, extrachargeexgst
							, totalexgst
							, totalgst
							, total
							, createdate
							, refusercreatedby
							, updatedate
							, refuser
							, description
							, notes
							, itemdate
							, fromlocation
							, tolocation
							, extrachargedescription
							)
						values (
							  invoiceid
							, docketrecord.refbooking
							, docketrecord.id
							, docketrecord.refdriver
							, docketrecord.reference1
							, docketrecord.refsuburbfrom
							, docketrecord.refsuburbto
							, 0::money
							, jobchargerecord.charge::money
							, coalesce(jobchargerecord.charge, 0::money)	--totalexgst
							, (case when ( fleetownerrecord.gstfree = 0) then 	 (coalesce(jobchargerecord.charge::numeric, 0::numeric)* v_gstvalue::numeric/ 100.00::numeric) else 0::numeric end)::money-- totalgst
							, (coalesce(jobchargerecord.charge::numeric, 0::numeric) + case when ( fleetownerrecord.gstfree = 0) then 	 (coalesce(jobchargerecord.charge::numeric, 0::numeric)* v_gstvalue::numeric/ 100.00::numeric)::numeric else 0::numeric end)::money
							, now()
							, p_refuser
							, now()
							, p_refuser
							, jobchargerecord.code
							, 'extra charge'
							, jobchargerecord.createdate
							, docketrecord.startlocationaddress
							, docketrecord.endlocationaddress
							, jobchargerecord.chargedescription
						);					
					end loop;  -- job deduction record--
					*/
					update docket set refdocketstatus = v_docketstatusrctiissued where id = docketrecord.id;
				end if;
			
			end loop;  --docketrecord--
			--end if; 
 	         	   
			totalexgst := coalesce(baserateexgst, 0::money) + extrachargeexgst;
			v_TotalForPeriodicCharges :=  totalexgst::numeric;
		
			if (p_generateinvoice = 1 and v_invoicepersisted = 1 and totalexgst > 0.0::money) then
 				for periodicchargerecord in 
 					select gc.refcharge refcharge
 							, cc.code calculationtype
							, gc.drivervalue drivervalue
							, coalesce(gc.addtototal, 0)  addtototal
							, ch.code chargecode
							, ch.description chargedescription
					from driverchargegroup_charge gc 
					inner join charge ch on ch.id = gc.refcharge 
					inner join  chargecalculation cc on cc.id = ch.refchargecalculation
					where ch.refchargetype = v_periodicchargetype 
						and gc.refdriverchargegroup = fleetownerrecord.refdriverchargegroup   
						--and gc.drivervalue > 0.0::numeric
						and gc.isactive = 1 
						and ch.isactive = 1
					order by gc.sequencenumber 
 				loop --periodicchargerecord--
	 				
	 			    v_PeriodicChargeValue := calculateCharge(periodicchargerecord.calculationtype, 1, v_TotalForPeriodicCharges, periodicchargerecord.drivervalue);
	 			   
	 			    if (v_PeriodicChargeValue > 0.0) then
	 			    	if (totalexgst > 0.0::money) then
	 						insert into rcti_invoice_charge (
								  refrctiinvoice
								, refcharge
								, qty
								, drivervalue
								, totalcharge
								, createdate
								, refusercreatedby
								, updatedate
								, refuser
								, isactive
								, chargecode
								, chargedescription
								, gst
								, chargedate
								)
							values (
								  invoiceid
								, periodicchargerecord.refcharge
								, 1
								, periodicchargerecord.drivervalue
								, v_PeriodicChargeValue
								, newInvoiceRecord.invoicedate
								, p_refuser
								, newInvoiceRecord.invoicedate
								, p_refuser
								, 1 
								, periodicchargerecord.chargecode
								, periodicchargerecord.chargedescription
								, case when (fleetownerrecord.gstfree = 0) then v_PeriodicChargeValue * v_gstvalue::numeric/ 100.00::numeric else 0::numeric end ::money -- totalgst
								, newInvoiceRecord.invoicedate						
							);			
							v_Rcti_InvoiceChargeExists := 1;
							totalexgst := totalexgst::numeric + v_PeriodicChargeValue::numeric;
						end if;
					end if;
				
	 			    if (v_PeriodicChargeValue < 0.0) then   --Negative periodic charge. create a separate invoice.
						if (v_DeductionInvoiceId = 0) then  -- create the deduction invoice in RCTI table 
							v_DeductionInvoiceId := nextval('RCTI_INVOICE_ID_SEQ');
							v_DeductionInvoiceTotalExGst := 0.0;
							v_DeductionInvoiceTotalGst  := 0.0;

							insert into rcti_invoice (
								  id
							  	, invoiceno			    
								, reffleetowner
								, refbranch
								, accountno
								, fleetownername
								, gstvalue
								, datefrom
								, dateto
								, refinvoicestatus
								, invoicestatusdate
								, createdate
								, refusercreatedby
								, updatedate
								, refuser
								, refdebitdocumenttype
								, creditmultiplier
							)
							values (
								  v_DeductionInvoiceId
								, invoiceno ||'RD'
								, reffleetowner
								, refbranch 
								, accountno 
								, fleetownername
								, gstvalue 
								, datefrom 
								, dateto 
								, refinvoicestatus 
								, invoicestatusdate 
								, createdate 
								, refusercreatedby 
								, updatedate 
								, refuser
								, v_debitdocumentDeductionTaxInvoice
								, -1::integer
						   );		   
						end if;
						--add deduction item into deduction invoice
	 					insert into rcti_invoice_charge(
	 						  refrctiinvoice
							, refcharge
							, qty
							, drivervalue
							, totalcharge
							, createdate
							, refusercreatedby
							, updatedate
							, refuser
							, isactive
							, chargecode
							, chargedescription
							, gst
							, chargedate  
						)
						values (							  
							  v_DeductionInvoiceId
							, periodicchargerecord.refcharge
							, 1
							, periodicchargerecord.drivervalue * -1
							, (v_PeriodicChargeValue * -1) ::money
							, newInvoiceRecord.invoicedate
							, p_refuser
							, newInvoiceRecord.invoicedate
							, p_refuser
							, 1 
							, periodicchargerecord.chargecode
							, periodicchargerecord.chargedescription
							, case when (fleetownerrecord.gstfree = 0) then (v_PeriodicChargeValue * -1) * v_gstvalue::numeric/ 100.00::numeric else 0::numeric end ::money -- totalgst
							, newInvoiceRecord.invoicedate						
						);			
						v_DeductionInvoiceTotalExGst := v_DeductionInvoiceTotalExGst + (v_PeriodicChargeValue * -1);

					end if; 
				
					if (periodicchargerecord.addtototal = 1) then
						v_TotalForPeriodicCharges := v_TotalForPeriodicCharges + v_PeriodicChargeValue;				
					end if;
				
				end loop; --periodicchargerecord-- 
			end if;

			---extrachargeexgst := coalesce(extrachargeexgst, 0::money) + coalesce(v_invoicechargevalue::money, 0::money);
			--totalexgst := coalesce(baserateexgst, 0::money) + extrachargeexgst;
			 
			if( fleetownerrecord.gstfree = 0) then 
				totalgst := (totalexgst * gstvalue::numeric)/100.00::numeric;
			end if;
			total := coalesce(totalexgst, 0::money) + coalesce(totalgst, 0::money);

			if (p_generateinvoice = 1  and v_invoicepersisted = 1 and v_rcti_invoiceitemexists = 1) then 
				v_baserateexgst := baserateexgst; 
				v_extrachargeexgst := extrachargeexgst ;  
			    v_totalexgst := totalexgst;
				v_totalgst := totalgst; 
				v_total:= total;
				update rcti_invoice set
				  	  baserateexgst = v_baserateexgst
					, extrachargeexgst = v_extrachargeexgst
					, totalexgst = v_totalexgst
					, totalgst = v_totalgst
					, total = v_total
					, totalitems = v_completedtotaldocketitems
					--, invoiceno =  nextval('INVOICE_NUMBER_SEQ')::varchar || 'R'
				where id = invoiceid; 
			end if;

			if (p_generateinvoice = 1  and v_invoicepersisted = 1 and v_DeductionInvoiceId > 0) then 
				update rcti_invoice set
				  	  baserateexgst = 0.0
					, extrachargeexgst = v_DeductionInvoiceTotalExGst ::money
					, totalexgst = v_DeductionInvoiceTotalExGst ::money
					, totalgst = case when (fleetownerrecord.gstfree = 0) then v_DeductionInvoiceTotalExGst::numeric * v_gstvalue::numeric/ 100.00::numeric else 0::numeric end ::money -- totalgst
					, total = v_DeductionInvoiceTotalExGst::money + case when (fleetownerrecord.gstfree = 0) then v_DeductionInvoiceTotalExGst * v_gstvalue::numeric/ 100.00::numeric else 0::numeric end ::money
					, totalitems = 0
				where id = v_DeductionInvoiceId; 
			end if;
			
			if (v_rcti_invoiceitemexists = 0 and v_invoicepersisted = 1 and v_Rcti_InvoiceChargeExists = 0) then 
					
					delete from rcti_invoice  where id = invoiceid;
					continue;
			end if; 

			if ( v_invoicepersisted = 0 and coalesce(baserateexgst::numeric,0.0) = 0 ) then 
					continue;
			end if; 
		  
	
			return next;  -- return current row of select
		end loop; --newInvoiceRecord--
	end loop;  --fleetownerrecord--
end;
